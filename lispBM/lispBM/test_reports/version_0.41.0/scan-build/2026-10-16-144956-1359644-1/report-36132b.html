540static bool_Bool generate_freelist(size_t num_cells) {
541 size_t i = 0;
542
543 if (!lbm_heap_state.heap) return false0;
544
545 lbm_heap_state.freelist = lbm_enc_cons_ptr(0);
546
547 lbm_cons_t *t;
548
549 // Add all cells to free list
550 for (i = 1; i < num_cells; i ++) {
551 t = lbm_ref_cell(lbm_enc_cons_ptr(i-1));
552 t->car = ENC_SYM_RECOVERED(((0x28) << 4) | 0x00000000u); // all cars in free list are "RECOVERED"
553 t->cdr = lbm_enc_cons_ptr(i);
554 }
555
556 // Replace the incorrect pointer at the last cell.
557 t = lbm_ref_cell(lbm_enc_cons_ptr(num_cells-1));
558 t->car = ENC_SYM_RECOVERED(((0x28) << 4) | 0x00000000u);
559 t->cdr = ENC_SYM_NIL(((0x0) << 4) | 0x00000000u);
560
561 return true1;
562}

707// ////////////////////////////////////////////////////////////
708// Deutch-Schorr-Waite (DSW) pointer reversal GC
709//
710/*
711 Deutch-Schorr-Waite (DSW) pointer reversal GC for 2-ptr cells
712 with a hack-solution for the lisp-array case (n-ptr cells).
713
714 DSW visits each branch node 3 times compared to 2 times for
715 the stack based recursive mark.
716 Where the stack based recursive mark performs a stack push/pop,
717 DSW rearranges the, current, prev, next and a ptr field on
718 the heap.
719
720 DSW changes the structure of the heap and it introduces an
721 invalid pointer (LBM_PTR_NULL) temporarily during marking.
722 Since the heap will be "messed up" while marking, a mutex
723 is introuded to keep other processes out of the heap while
724 marking.
725*/
726
727static int do_nothing(lbm_value v, bool_Bool shared, void *arg) {
728 (void) v;
729 (void) shared;
730 (void) arg;
731 return TRAV_FUN_SUBTREE_CONTINUE1;
732}

734void lbm_gc_mark_phase_ptr_rev(lbm_value root) {
735 lbm_mutex_lock(&lbm_const_heap_mutex);
736 lbm_ptr_rev_trav(do_nothing, root, NULL((void*)0));
737 lbm_mutex_unlock(&lbm_const_heap_mutex);
738}

741// Check if a value is currently on the stack

743bool_Bool active_ptr(lbm_value p) {
744 lbm_stack_t *s = &lbm_heap_state.gc_stack;
745 bool_Bool r = false0;
746 for (lbm_uint i = 0; i < s->sp; i ++) {
747 if (p == s->data[i]) {
748 r = true1;
749 break;
750 }
751 }
752 return r;
753}

755// ////////////////////////////////////////////////////////////
756// Explicit stack "recursive" mark phase
757
758// Trees are marked in a left subtree before rigth subtree, car first then cdr,
759// way to favor lisp lists. This means that stack will grow slowly when
760// marking right-leaning (cdr-recursive) data-structures while left-leaning
761// (car-recursive) structures uses a lot of stack.
762
763// Lisp arrays contain an extra book-keeping field to keep track
764// of how far into the array the marking process has gone.
765
766// TODO: DSW should be used as a last-resort if the GC stack is exhausted.
767// If we use DSW as last-resort can we get away with a way smaller
768// GC stack and unchanged performance (on sensible programs)?
769
770void lbm_gc_mark_phase(lbm_value root) {
771 lbm_value t_ptr;
772 lbm_stack_t *s = &lbm_heap_state.gc_stack;
773 s->data[s->sp++] = root;
774
775 while (!lbm_stack_is_empty(s)) {
776 lbm_value curr;
777 lbm_pop(s, &curr);
778
779 mark_shortcut:
780
781 if (!lbm_is_ptr(curr) ||
782 (curr & LBM_PTR_TO_CONSTANT_BIT0x04000000u)) {
783 continue;
784 }
785
786 lbm_cons_t *cell = &lbm_heap_state.heap[lbm_dec_ptr(curr)];
787
788 if (lbm_get_gc_mark(cell->cdr)) {
789 continue;
790 }
791
792 t_ptr = lbm_type_of(curr);
793
794 // An array is marked in O(N) time using an additional 32bit
795 // value per array that keeps track of how far into the array GC
796 // has progressed.
797 if (t_ptr == LBM_TYPE_LISPARRAY0xB0000000u) {
798 // O(stack_size) cost when seeing array. This protects
799 // against gc recursing into arrays via cycles.
800 if (active_ptr(curr)) {
801 continue; // Already in process of marking this array, abort cycle!
802 }
803 lbm_array_header_extended_t *arr = (lbm_array_header_extended_t*)cell->car;
804 lbm_value *arrdata = (lbm_value *)arr->data;
805 if (!lbm_push(s, curr)) { // put array back as bookkeeping.
806 // Switch to pointer reversal here.
807 // This is a new array that we have not yet started descending into.
808 lbm_gc_mark_phase_ptr_rev(curr);
809 continue;
810 }
811 // Example A: Array with 10 elements
812 // A: assume arr->index == 9
813 // Example B: Array with 0 elements
814 // B: assume arr->index == 0
815 if (arr->index < ((arr->size/(sizeof(lbm_value))))) { // A: 9 < 10
816 // A: True path
817 // B: 0 < 0
818 // B: False path
819 if (lbm_is_ptr(arrdata[arr->index]) && ((arrdata[arr->index] & LBM_PTR_TO_CONSTANT_BIT0x04000000u) == 0) &&
820 !((arrdata[arr->index] & LBM_CONTINUATION_INTERNAL0xF8000001u) == LBM_CONTINUATION_INTERNAL0xF8000001u)) {
821
822 lbm_cons_t *elt = &lbm_heap_state.heap[lbm_dec_ptr(arrdata[arr->index])];
823 if (!lbm_get_gc_mark(elt->cdr)) {
824 curr = arrdata[arr->index];
825 arr->index++;
826 goto mark_shortcut;
827 }
828 }
829 // We need one more round of the loop if there is at least one element left in the array.
830 // If we skip this check things should work the same, but there will be one more recurse
831 // before we discover in the earlier conditional that we have reached the end of the array.
832 if (arr->index < ((arr->size/(sizeof(lbm_value))) - 1)) { // A: 9 < 9
833 // A: False path
834 arr->index++;
835 continue;
836 }
837 }
838 arr->index = 0;
839 lbm_stack_drop(s,1); // Drop the bookkeeping element.
840 // elem_dropped == cell
841 // Cell is the array reference. so this marks the array reference itself
842 // after having marked all the elements.
843 cell->cdr = lbm_set_gc_mark(cell->cdr);
844 lbm_heap_state.gc_marked ++;
845 continue;
846 } else if (t_ptr == LBM_TYPE_CHANNEL0x90000000u) {
847 cell->cdr = lbm_set_gc_mark(cell->cdr);
848 lbm_heap_state.gc_marked ++;
849 // TODO: Can channels be explicitly freed ?
850 if (cell->car != ENC_SYM_NIL(((0x0) << 4) | 0x00000000u)) {
851 lbm_char_channel_t *chan = (lbm_char_channel_t *)cell->car;
852 // A channel dependency is currently assumed to be a byte array (a string)
853 // or NIL.
854 curr = chan->dependency;
855 goto mark_shortcut;
856 }
857 continue;
858 }
859
860 cell->cdr = lbm_set_gc_mark(cell->cdr);
861 lbm_heap_state.gc_marked ++;
862
863 if (t_ptr == LBM_TYPE_CONS0x10000000u) {
864 if (lbm_is_ptr(cell->cdr)) {
865 if (!lbm_push(s, cell->cdr)) {
866 // This is the only place where we grab a new stack position.
867 // If the stack is full switch to ptr-rev GC.
868 lbm_gc_mark_phase_ptr_rev(cell->cdr);
869 }
870 }

921// Sweep moves non-marked heap objects to the free list.
922int lbm_gc_sweep_phase(void) {
923 unsigned int i = 0;
924 lbm_cons_t *heap = (lbm_cons_t *)lbm_heap_state.heap;
925
926 for (i = 0; i < lbm_heap_state.heap_size; i ++) {
927 if ( lbm_get_gc_mark(heap[i].cdr)) {
928 heap[i].cdr = lbm_clr_gc_mark(heap[i].cdr);
929 } else {
930 // Check if this cell is a pointer to an array
931 // and free it.
932 if (lbm_type_of(heap[i].cdr) == LBM_TYPE_SYMBOL0x00000000u) {
933 switch(heap[i].cdr) {
934
935 case ENC_SYM_IND_I_TYPE(((0x34) << 4) | 0x00000000u): /* fall through */
936 case ENC_SYM_IND_U_TYPE(((0x35) << 4) | 0x00000000u):
937 case ENC_SYM_IND_F_TYPE(((0x36) << 4) | 0x00000000u):
938 lbm_memory_free((lbm_uint*)heap[i].car);
939 break;
940 case ENC_SYM_DEFRAG_LISPARRAY_TYPE(((0x3C) << 4) | 0x00000000u): /* fall through */
941 case ENC_SYM_DEFRAG_ARRAY_TYPE(((0x3B) << 4) | 0x00000000u):
942 lbm_defrag_mem_free((lbm_uint*)heap[i].car);
943 break;
944 case ENC_SYM_LISPARRAY_TYPE(((0x39) << 4) | 0x00000000u): /* fall through */
945 case ENC_SYM_ARRAY_TYPE(((0x30) << 4) | 0x00000000u):{
946 lbm_array_header_t *arr = (lbm_array_header_t*)heap[i].car;
947 lbm_memory_free((lbm_uint *)arr->data);
948 lbm_heap_state.gc_recovered_arrays++;
949 lbm_memory_free((lbm_uint *)arr);
950 } break;
951 case ENC_SYM_CHANNEL_TYPE(((0x37) << 4) | 0x00000000u):{
952 lbm_char_channel_t *chan = (lbm_char_channel_t*)heap[i].car;
953 lbm_memory_free((lbm_uint*)chan->state);
954 lbm_memory_free((lbm_uint*)chan);
955 } break;
956 case ENC_SYM_CUSTOM_TYPE(((0x38) << 4) | 0x00000000u): {
957 lbm_uint *t = (lbm_uint*)heap[i].car;
958 lbm_custom_type_destroy(t);
959 lbm_memory_free(t);
960 } break;
961 case ENC_SYM_DEFRAG_MEM_TYPE(((0x3A) << 4) | 0x00000000u): {
962 lbm_uint *ptr = (lbm_uint *)heap[i].car;
963 lbm_defrag_mem_destroy(ptr);
964 } break;
965 default:
966 break;
967 }
968 }
969 // create pointer to use as new freelist
970 lbm_uint addr = lbm_enc_cons_ptr(i);
971
972 // Clear the "freed" cell.
973 heap[i].car = ENC_SYM_RECOVERED(((0x28) << 4) | 0x00000000u);
974 heap[i].cdr = lbm_heap_state.freelist;
975 lbm_heap_state.freelist = addr;
976 lbm_heap_state.num_free++;
977 lbm_heap_state.gc_recovered ++;
978 }
979 }
980 return 1;
981}

1415// ////////////////////////////////////////////////////////////
1416// pointer reversal traversal
1417//
1418// Caveats:
1419// * Structures on the constant heap cannot be traversed using
1420// pointer reversal. If a dynamic structure is pointing into the
1421// constant heap, the 'f' will be applied to the constant cons cell on
1422// the border and then traversal will retreat.
1423//
1424// * It may be impossible to detect all kinds of cycles
1425// if attempting to use and restore the GC in a single pass
1426// over the tree. To detect cycles all visited nodes must
1427// remain detectable when traversing all branches!
1428//
1429// * Potential fix is to run GC after a complete traversal of the
1430// entire value in order to restore the GC bits.
1431//
1432// * If we leave GC bits set when traversing values, we can use this
1433// to detect cycles that happen in multiple steps accross values
1434// in the environment.
1435//
1436// * lbm_ptr_rev_trav with the "do_nothing" travfun is the same thing
1437// as a GC mark phase! Maybe utilize this for code-size
1438// purposes. This also increases the amount of testing the
1439// ptr_rev_trav function is subjected to.
1440
1441void lbm_ptr_rev_trav(trav_fun f, lbm_value v, void* arg) {
1442
1443 lbm_value curr = v;
1444 lbm_value prev = lbm_enc_cons_ptr(LBM_PTR_NULL(0x03FFFFFCu >> 2));
1445 while (true1) {
1446
1447 // Run leftwards and process conses until
1448 // hitting a leaf in the left direction.
1449
1450 // If curr is marked here there is a cycle in the graph.
1451 // In case of a cycle or leaf, this first loop is exited.
1452 while (((lbm_is_cons_rw(curr)) ||
1453 (lbm_is_lisp_array_rw(curr)) ||
1454 (lbm_is_channel(curr))) && !gc_marked(curr)) {
1455 lbm_cons_t *cell = lbm_ref_cell(curr);
1456 if (lbm_is_cons(curr)) {
1457 // In-order traversal
1458 if (f(curr, false0, arg) == TRAV_FUN_SUBTREE_DONE0) {
1459 lbm_gc_mark_phase(curr);
1460 goto trav_backtrack;
1461 }
1462 gc_mark(curr);
1463
1464 lbm_value next = 0;
1465 value_assign(&next, cell->car);
1466 value_assign(&cell->car, prev);
1467 value_assign(&prev, curr);
1468 value_assign(&curr, next);
1469 } else if (lbm_is_channel (curr)) {
1470 if (f(curr, false0, arg) == TRAV_FUN_SUBTREE_DONE0) {
1471 lbm_gc_mark_phase(curr);
1472 goto trav_backtrack;
1473 }
1474 gc_mark(curr);
1475
1476 lbm_char_channel_t *chan = (lbm_char_channel_t*)cell->car;
1477 lbm_value next = 0;
1478 value_assign(&next, chan->dependency);
1479 value_assign(&chan->dependency, prev);
1480 value_assign(&prev, curr);
1481 value_assign(&curr, next);
1482 } else { // it is an array
1483 lbm_array_header_extended_t *arr = (lbm_array_header_extended_t*)cell->car;
1484 lbm_value *arr_data = (lbm_value *)arr->data;
1485 if (arr->size == 0) break;
1486
1487 if (f(curr, false0, arg) == TRAV_FUN_SUBTREE_DONE0) {
1488 lbm_gc_mark_phase(curr);
1489 break;
1490 }
1491 // Cycles in arrays are not detected using the index
1492 // in this algorithm. If we enter into an array, we take
1493 // full ownership of it here, even if the stack based marker
1494 // has partially walked it already.
1495 arr->index = 1; // Explicitly sets array index to one.
1496 gc_mark(curr);
1497
1498 lbm_value next = 0;
1499 value_assign(&next, arr_data[0]);
1500 value_assign(&arr_data[0], prev);
1501 value_assign(&prev, curr);
1502 value_assign(&curr, next);
1503 }
1504 }
1505 // Currently there are a few different users of this traversal.
1506 // size, flatten and detect_sharing.
1507 // detect_sharing make use of the shared (true) argument in f(curr, true, arg)
1508 // while the other do not. detect_sharing also assumes it is run once per env item
1509 // while not resetting any GC-bits in between. This detects global sharing.
1510
1511 if (lbm_is_ptr(curr) && gc_marked(curr)) {
1512 // gc bit set so this subtree is already traversed.
1513 // f is called with true to indicate visited node.
1514 // if this happens during a sharing discovery phase, curr will be added to sharing table.
1515 f(curr, true1, arg);
1516 // In this case f should not be able to return subtree continue.
1517 // The only correct return value from f is PROCEED.
1518 } else if (!lbm_is_cons(curr) || // Found a leaf
1519 (curr & LBM_PTR_TO_CONSTANT_BIT0x04000000u)) {
1520 if (lbm_is_ptr(curr) && !(curr & LBM_PTR_TO_CONSTANT_BIT0x04000000u)) gc_mark(curr); // Mark it so that the mandatory GC does not swipe it.
1521 f(curr, false0, arg);
1522 }
1523
1524 // Now either prev has the "flag" set or it doesnt.
1525 // If the flag is set that means that the prev node
1526 // have had both its car and cdr visited. So that node is done!
1527 //
1528 // If the flag is not set, jump down to SWAP
1529
1530 trav_backtrack:
1531 while ((lbm_is_cons(prev) &&
1532 (lbm_dec_ptr(prev) != LBM_PTR_NULL(0x03FFFFFCu >> 2)) && // is LBM_NULL a cons type?
1533 lbm_get_gc_flag(lbm_car(prev))) ||
1534 lbm_is_lisp_array_rw(prev) ||
1535 lbm_is_channel(prev)) {
1536 lbm_cons_t *cell = lbm_ref_cell(prev);
1537 if (lbm_is_cons(prev)) {
1538
1539 // clear the flag
1540 // This means that we are done with a "CDR" child.
1541 // prev = [ a , b ][flag = 1]
1542 // =>
1543 // prev = [ a , b ][flag = 0]
1544
1545 //gc_clear_mark(prev);
1546 cell->car = lbm_clr_gc_flag(cell->car);
1547 // Move on downwards until
1548 // finding a cons cell without flag or NULL
1549
1550 // curr = c
1551 // prev = [ a , b ][flag = 0]
1552 // =>
1553 // prev = [ a , c ][flag = 0]
1554 // curr = prev
1555 // prev = b
1556
1557 lbm_value next = 0;
1558 value_assign(&next, cell->cdr);
1559 value_assign(&cell->cdr, curr);
1560 value_assign(&curr, prev);
1561 value_assign(&prev, next);
1562 } else if (lbm_is_channel(prev)) {
1563 lbm_char_channel_t *chan = (lbm_char_channel_t*)cell->car;
1564 lbm_value next = 0;
1565 value_assign(&next, chan->dependency);
1566 value_assign(&chan->dependency, curr);
1567 value_assign(&curr, prev);
1568 value_assign(&prev, next);
1569 } else { // is an array
1570 lbm_array_header_extended_t *arr = (lbm_array_header_extended_t*)cell->car;
1571 lbm_value *arr_data = (lbm_value *)arr->data;
1572 size_t arr_size = (size_t)arr->size / sizeof(lbm_value);
1573 lbm_value next = 0;
1574 if (arr->index == arr_size) {
1575 value_assign(&next, arr_data[arr->index-1]);
1576 value_assign(&arr_data[arr->index-1], curr);
1577 value_assign(&curr, prev);
1578 value_assign(&prev, next);
1579 arr->index = 0;
1580 } else {
1581 break;
1582 }
1583 }
1584 }
1585
1586 // SWAP
1587
1588 // if the prev node is NULL we have traced backwards all the
1589 // way back to where curr == v. Another alternative is that
1590 // the input v was an Atom. We are done!
1591 if (lbm_is_ptr(prev) &&
1592 lbm_dec_ptr(prev) == LBM_PTR_NULL(0x03FFFFFCu >> 2)) {
1593 if (lbm_is_cons(curr)) {
1594 //gc_clear_mark(curr);
1595 }
1596 //done = true;
1597 break;
1598 }
1599
1600 // if the prev node is not NULL then we should move
1601 // down to the prev node and start process its remaining child.
1602 else if (lbm_is_cons(prev)) {
1603
1604 lbm_cons_t *cell = lbm_ref_cell(prev);
1605 lbm_value next = 0;
1606
1607
1608 // prev = [ p , cdr ][flag = 0]
1609 // =>
1610 // prev = [ p , cdr ][flag = 1]
1611
1612 cell->car = lbm_set_gc_flag(cell->car);
1613
1614 // switch to processing the cdr field and set the flag.
1615 // curr = c
1616 // prev = [ a, b ][flag = 1]
1617 // =>
1618 // prev = [ c, a ][flag = 1]
1619 // curr = b
1620
1621 value_assign(&next, cell->car);
1622 value_assign(&cell->car, curr);
1623 value_assign(&curr, cell->cdr);
1624 value_assign(&cell->cdr, next);
1625 } else if (lbm_is_lisp_array_rw(prev)) {
1626 lbm_cons_t *cell = lbm_ref_cell(prev);
1627 lbm_array_header_extended_t *arr = (lbm_array_header_extended_t*)cell->car;
1628 lbm_value *arr_data = (lbm_value *)arr->data;
1629 lbm_value next = 0;
1630
1631 value_assign(&next, arr_data[arr->index-1]);
1632 value_assign(&arr_data[arr->index-1], curr);
1633 value_assign(&curr, arr_data[arr->index]);
1634 value_assign(&arr_data[arr->index], next);
1635 arr->index = arr->index + 1;
1636 }
1637 }
1638}
