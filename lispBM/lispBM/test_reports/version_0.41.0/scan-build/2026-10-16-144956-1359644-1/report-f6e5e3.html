313static lbm_uint float_to_u(float number) {
314 // Set subnormal numbers to 0 as they are not handled properly
315 // using this method.
316 if (fabsf(number) < 1.5e-38) {
317 number = 0.0;
318 }
319
320 int e = 0;
321 float sig = frexpf(number, &e);
322 float sig_abs = fabsf(sig);
323 uint32_t sig_i = 0;
324
325 if (sig_abs >= 0.5) {
326 sig_i = (uint32_t)((sig_abs - 0.5f) * 2.0f * 8388608.0f);
327 e += 126;
328 }

330 uint32_t res = (((uint32_t)e & 0xFFu) << 23) | (uint32_t)(sig_i & 0x7FFFFFu);
331 if (sig < 0) {
